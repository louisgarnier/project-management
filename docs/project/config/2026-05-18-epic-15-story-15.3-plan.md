> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` to implement this plan task-by-task.

```tsx
"use client";

import { useState } from "react";

type Props = {
  terms: string[];
  editable?: boolean;
  onChange?: (next: string[]) => void;  // required when editable=true
};

export default function KeyTermChips({ terms, editable = false, onChange }: Props) {
  const [draft, setDraft] = useState("");

  const remove = (i: number) => {
    onChange?.(terms.filter((_, idx) => idx !== i));
  };
  const add = () => {
    const v = draft.trim();
    if (!v || terms.includes(v)) {
      setDraft("");
      return;
    }
    onChange?.([...terms, v]);
    setDraft("");
  };

  return (
    <div style={{ display: "flex", flexWrap: "wrap", gap: 3, alignItems: "center" }}>
      {terms.map((t, i) => (
        <span
          key={`${t}-${i}`}
          style={{
            fontSize: 10.5,
            background: "#f4f5f7",
            border: "1px solid #dfe1e6",
            color: "#42526e",
            padding: "2px 8px",
            borderRadius: 10,
            display: "inline-flex",
            alignItems: "center",
            gap: 4,
          }}
        >
          {t}
          {editable && (
            <button
              type="button"
              onClick={() => remove(i)}
              style={{
                border: "none",
                background: "none",
                cursor: "pointer",
                color: "#97a0af",
                fontSize: 10,
                padding: 0,
                lineHeight: 1,
              }}
              aria-label={`remove ${t}`}
            >
              ×
            </button>
          )}
        </span>
      ))}
      {editable && (
        <>
          <input
            value={draft}
            onChange={(e) => setDraft(e.target.value)}
            onKeyDown={(e) => {
              if (e.key === "Enter") {
                e.preventDefault();
                add();
              }
            }}
            placeholder="+ term"
            style={{
              fontSize: 10.5,
              border: "1px dashed #c1c7d0",
              background: "transparent",
              borderRadius: 10,
              padding: "1px 8px",
              width: 80,
              outline: "none",
            }}
          />
        </>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { callsAPI, libraryAPI, topicsAPI } from "@/api/client";
import { logger } from "@/utils/logger";
import type { Call, TopicData, TaskData, EvidenceRef, LibraryEntry } from "@/types";
import KeyTermChips from "./KeyTermChips";
import EvidenceRefPopover from "./EvidenceRefPopover";

type Props = {
  call: Call;
  onAggregateComplete: () => void;
  onAutoAdvanced: () => void;
  onPollCall?: () => Promise<void>;
};

const STATUS_BG: Record<TaskData["status"], string> = {
  open: "#e9f0ff",
  in_progress: "#fff4e6",
  resolved: "#e3fcef",

const STATUS_FG: Record<TaskData["status"], string> = {
  open: "#0052cc",
  in_progress: "#974f0c",
  resolved: "#006644",

const IMP_BG: Record<TopicData["importance"], string> = {
  high: "#ffebe5",
  medium: "#fffae6",
  low: "#f4f5f7",
};
const IMP_FG: Record<TopicData["importance"], string> = {
  high: "#bf2600",
  medium: "#974f0c",
  low: "#5e6c84",

  const [topics, setTopics] = useState<TopicData[]>([]);
  const [libraryPrompts, setLibraryPrompts] = useState<LibraryEntry[]>([]);
  const [selectedPromptId, setSelectedPromptId] = useState<string | null>(call.call_topics_prompt_id ?? null);
  const [loading, setLoading] = useState(true);

  // ── Load library prompts for the selector ──
  useEffect(() => {
    libraryAPI.listByCategory("call_topics").then(setLibraryPrompts).catch((e) => {
      logger.error("[CallTopicsStage] library load failed", e);
    });
  }, []);

  // ── Load topics for this call ──
  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    topicsAPI.listForCall(call.id).then((data) => {
      if (cancelled) return;
      setTopics(data);
      setLoading(false);
    }).catch((e) => {
      logger.error("[CallTopicsStage] topics load failed", e);
      setLoading(false);
    });
    return () => { cancelled = true; };
  }, [call.id]);

  // Debounced patch helper
  const patchTopic = async (idx: number, partial: Partial<TopicData>) => {
    const topic = topics[idx];
    if (!topic.id) return;
    setTopics((prev) => prev.map((t, i) => (i === idx ? { ...t, ...partial } : t)));
    try {
      await topicsAPI.patch(topic.id, partial as Record<string, unknown>);
    } catch (e) {
      logger.error("[CallTopicsStage] patch topic failed", e);
    }
  };

  // Tasks: rebuild full array + send
  const updateTasks = async (idx: number, newTasks: TaskData[]) => {
    setTopics((prev) => prev.map((t, i) => (i === idx ? { ...t, tasks: newTasks } : t)));
    const topic = topics[idx];
    if (!topic.id) return;
    try {
      await topicsAPI.patch(topic.id, { tasks: newTasks });
    } catch (e) {
      logger.error("[CallTopicsStage] tasks patch failed", e);
    }
  };

  const deleteTopic = async (idx: number) => {
    const topic = topics[idx];
    if (!topic.id) return;
    if (!confirm(`Delete topic "${topic.name}"?`)) return;
    setTopics((prev) => prev.filter((_, i) => i !== idx));
    try {
      await topicsAPI.deleteTopic(topic.id);
    } catch (e) {
      logger.error("[CallTopicsStage] delete topic failed", e);
    }
  };

  const onPromptChange = async (id: string | null) => {
    setSelectedPromptId(id);
    try {
      await callsAPI.patchPromptSelection(call.id, id);
    } catch (e) {
      logger.error("[CallTopicsStage] prompt selection failed", e);
    }
  };

  const taskCount = topics.reduce((n, t) => n + t.tasks.length, 0);

  return (
    <div style={{ padding: 24, background: "#f4f5f7" }}>
      {/* Header */}
      <div style={{ fontSize: 11, color: "#5e6c84", textTransform: "uppercase", letterSpacing: ".06em", marginBottom: 6 }}>
        Call Topics · {call.title || "Untitled"} · {(call.created_at || "").slice(0, 10)}
      </div>
      <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: 18 }}>
        <h2 style={{ fontSize: 18, margin: 0 }}>Extracted tasks ({taskCount} tasks across {topics.length} topics)</h2>
        <div>
          <label style={{ fontSize: 11, color: "#5e6c84", marginRight: 8 }}>Prompt:</label>
          <select
            value={selectedPromptId ?? ""}
            onChange={(e) => onPromptChange(e.target.value || null)}
            style={{ fontSize: 12, padding: "4px 8px", border: "1px solid #dfe1e6", borderRadius: 4 }}
          >
            <option value="">(library default)</option>
            {libraryPrompts.map((p) => (
              <option key={p.id} value={p.id}>{p.name}</option>
            ))}
          </select>
        </div>
      </div>

      {/* Table */}
      {loading ? (
        <div style={{ color: "#5e6c84" }}>Loading topics…</div>
      ) : topics.length === 0 ? (
        <div style={{ color: "#5e6c84" }}>No topics yet. Run extraction first.</div>
      ) : (
        <div style={{ background: "white", border: "1px solid #dfe1e6", borderRadius: 6, overflow: "visible" }}>
          <table style={{ width: "100%", borderCollapse: "collapse", fontSize: 13, tableLayout: "fixed" }}>
            <colgroup>
              <col style={{ width: 300 }} />
              <col />
              <col style={{ width: 300 }} />
              <col style={{ width: 92 }} />
              <col style={{ width: 108 }} />
              <col style={{ width: 46 }} />
              <col style={{ width: 50 }} />
            </colgroup>
            <thead>
              <tr style={{ background: "#fafbfc", color: "#5e6c84", fontSize: 10, textTransform: "uppercase", letterSpacing: ".05em" }}>
                <th style={th}>Topic / key terms</th>
                <th style={th}>Task</th>
                <th style={th}>Next step</th>
                <th style={th}>Owner</th>
                <th style={th}>Status</th>
                <th style={{ ...th, textAlign: "center" }}>Ev.</th>
                <th style={{ ...th, textAlign: "center" }}></th>
              </tr>
            </thead>
            <tbody>
              {topics.flatMap((topic, ti) => {

                    {/* Topic / chips — repeated on every row per v3 */}

                          style={topicNameStyle}

                          style={{ ...impSelect, background: IMP_BG[topic.importance], color: IMP_FG[topic.importance] }}
```
