**Goal:** Ship the xlsx exporter (5 sheets via openpyxl), the `GET /api/projects/{id}/export.xlsx` endpoint, and the Project Tracker sub-tab inside the Artifacts page (5 sub-views per the approved v3 mockup).

```python
"""xlsx tracker exporter (Story 15.8).

build_tracker_xlsx(project_id) -> bytes: renders 5 sheets via openpyxl.
"""

from io import BytesIO
from datetime import datetime

from openpyxl import Workbook
from openpyxl.styles import Alignment, Font, PatternFill
from openpyxl.utils import get_column_letter

from backend.services.topics_service import list_project_topics, list_topics_timeline


def build_tracker_xlsx(project_id: str) -> bytes:
    """Render the 5-sheet tracker xlsx and return as bytes.

    Sheets:
    - Dashboard: high-level counts
    - Chronology: 1 row per topic × 1 col per call + RAG note
    - Anchors lifecycle: 1 row per task/open_question
    - Decisions log: 1 row per decision
    - Key terms registry: 1 row per (topic, key_term)
    """
    project_topics = list_project_topics(project_id)
    timeline = list_topics_timeline(project_id) if project_topics else None

    wb = Workbook()
    # Default 'Sheet' is auto-created; rename to first sheet
    wb.remove(wb.active)

    _build_dashboard(wb, project_topics)
    _build_chronology(wb, project_topics, timeline)
    _build_anchors_lifecycle(wb, project_topics)
    _build_decisions_log(wb, project_topics)
    _build_key_terms_registry(wb, project_topics)

    stream = BytesIO()
    wb.save(stream)
    return stream.getvalue()


# Internal helpers — _build_dashboard, _build_chronology, _build_anchors_lifecycle,
# _build_decisions_log, _build_key_terms_registry. Each takes the workbook and the
# data, creates a sheet, writes the header row + data rows. All use wrap_text + freeze_panes.
```

## Task 3: GET /api/projects/{id}/export.xlsx endpoint

Add to `backend/routers/projects.py`:

```python
from datetime import datetime, timezone
from fastapi.responses import StreamingResponse
from io import BytesIO

from backend.exporters.xlsx_tracker import build_tracker_xlsx
from backend.services.export_service import _slugify


@router.get("/projects/{project_id}/export.xlsx")
async def export_tracker_xlsx(project_id: str):
    """Streaming xlsx download for the project tracker (Story 15.8)."""
    db = get_client()
    rows = db.table("projects").select("name").eq("id", project_id).execute().data
    if not rows:
        raise HTTPException(status_code=404, detail=f"Project {project_id} not found")
    project_name = rows[0]["name"]
    blob = build_tracker_xlsx(project_id)
    slug = _slugify(project_name)
    iso_date = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    filename = f"{slug}-tracker-{iso_date}.xlsx"
    return StreamingResponse(
        BytesIO(blob),
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )
```
